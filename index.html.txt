<script>
const NOTES = {
  intro: "BÜCÜRRR İYKİ DOĞDUNN BAŞ TACISINN BUNU MUTLU OLMANI DİLEYEREK YAZIYORM  ❤️",
  tasks: [
    "Görev 1: İlk 2 KALBİ yakala! ❤️",
    "Görev 2: BİRAZ HIZLI OL BÜCÜR! 💨",
    "Görev 3: BUDA HEDİYE OLSUN HADİ! 🎁",
    "Görev 4: Şimdi en çok sevdiğin kalpleri yakala! 💖",
    "Görev 5: SON BABA BUNLAR! 🏁",
    "Görev 6: AFERİM BÜCÜR BAŞKA KALMADI MAALESEF SON BUNLAR! 🎉"
  ],
  taskCompletePrefix: "TEBRİKLERRR BÜCÜRRRRR! 🎉 ",
  finalTitle: "BÜCÜRRRR İYİKİ DOĞDUNNN İYİKİ VARSINNN BAİ TACISINNN 🎂",
  finalText: "MUTLU SENELEREEE EE ❤️",
  aferimText: "HELALL BÜCÜRRR ❤️"
};

const TOTAL=20;
const TASKS=[2,4,6,8,10,12];
let currentTask=0, collected=0, hearts=[];

const gameArea=document.getElementById("gameArea");
const message=document.getElementById("message");
const final=document.getElementById("final");
const bgm=document.getElementById("bgm");
const pop=document.getElementById("pop");
const win=document.getElementById("win");
const taskMsg=document.getElementById("taskMsg");

document.getElementById("girlPhoto").onclick = () => {
  let url = prompt("Fotoğraf URL’si gir:"); 
  if(url) document.getElementById("girlPhoto").src = url;
};

document.getElementById("storyText").onclick = () => {
  let text = prompt("Hikâyeyi yaz:"); 
  if(text) document.getElementById("storyText").innerText = text;
};

document.getElementById("startBtn").onclick=()=>{
  document.getElementById("intro").style.display="none";
  bgm.play();
  startGame();
};

function random(min,max){return Math.random()*(max-min)+min}

function spawnHeart(){
  const h=document.createElement("img");
  h.src="data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 24 24'><path fill='%23ff2d6f' d='M12 21s-7.5-4.86-10.18-8.02C-0.09 8.6 3.28 3 8.5 4.6 11 5.45 12 8 12 8s1-2.55 3.5-3.4C20.72 3 24.09 8.6 22.18 12.98 19.5 16.14 12 21 12 21z'/></svg>";
  h.className="heart";
  h.style.left=random(40,window.innerWidth-100)+"px";
  h.style.top=random(100,window.innerHeight-140)+"px";
  h.dataset.vx=random(-0.6,0.6); h.dataset.vy=random(-0.4,0.4);
  h.onclick=()=>{
    pop.play(); 
    h.remove(); 
    collected++;
    message.innerText=NOTES.aferimText; 
    message.style.display="block";
    setTimeout(()=>message.style.display="none",900);
    burstConfetti(parseFloat(h.style.left),parseFloat(h.style.top));
    checkTask();
  };
  gameArea.appendChild(h); 
  return h;
}

function burstConfetti(x,y){
  const colors=['#ff6aa6','#ffd166','#6fe7ff','#a9ffbf','#ffd1f5'];
  for(let i=0;i<12;i++){
    const c=document.createElement('div');c.className='confetti';
    c.style.left=x+'px'; c.style.top=y+'px'; c.style.background=colors[i%colors.length];
    document.body.appendChild(c);
    let vx=random(-8,8), vy=random(-18,-4), tt=0, rot=random(-600,600);
    let id=setInterval(()=>{
      tt+=16;
      c.style.left=(x+vx*(tt/16))+'px'; 
      c.style.top=(y+vy*(tt/16)+0.32*0.5*(tt/16*tt/16))+'px'; 
      c.style.opacity=Math.max(0,1-tt/900); 
      c.style.transform='rotate('+(rot*(tt/1000))+'deg)'; 
      if(tt>900){clearInterval(id);c.remove();}
    },16);
  }
}

function startGame(){for(let i=0;i<TOTAL;i++){setTimeout(()=>hearts.push(spawnHeart()),i*150);} animate();}

function animate(){hearts.forEach(h=>{if(!h.parentNode) return; let x=parseFloat(h.style.left),y=parseFloat(h.style.top); let vx=parseFloat(h.dataset.vx),vy=parseFloat(h.dataset.vy); x+=vx;y+=vy; if(x<0||x>window.innerWidth-80) h.dataset.vx=-vx; if(y<50||y>window.innerHeight-100) h.dataset.vy=-vy; h.style.left=x+'px'; h.style.top=y+'px';}); requestAnimationFrame(animate);}

function checkTask(){
  if(collected>=TASKS[currentTask]){
    taskMsg.innerText=NOTES.taskCompletePrefix+NOTES.tasks[currentTask];
    currentTask++;
    setTimeout(()=>{
      if(currentTask<TASKS.length){
        taskMsg.innerText=NOTES.tasks[currentTask];
      } else {
        win.play();
        final.style.display="block";
        taskMsg.style.display="none";
      }
    },1200);
  }
}
